def handle_billing(request: str) -> str:
    return f"Routed to billing: {request}"

def handle_technical(request: str) -> str:
    return f"Routed to technical support: {request}"

ROUTES = {"billing": handle_billing, "technical": handle_technical}

def normalize_request(request: str) -> str:
    return request.strip()

async def process_ticket(client, ticket: dict) -> str:
    await asyncio.sleep(0.3)
    normalized = normalize_request(ticket["request"])
    response = client.create(messages=[{"role": "user", "content": f"Classify this support request as exactly 'billing' or 'technical': {normalized}"}])
    category = response.content[0].text
    handler = ROUTES[category]
    return handler(normalized)

async def run_ticket_pipeline(client, tickets: list) -> str:
    results = await asyncio.gather(*(process_ticket(client, t) for t in tickets))
    return "\\n".join(results)`,
