export const PARSE = String.raw`import re
_p = prompt
_t = " ".join(_p.lower().split())
_blocks = sorted(
    [(m.start(), m.end(), m.group(2)) for m in re.finditer(r"<(\w+)>(.*?)</\1>", _p, re.S)]
    + [(m.start(), m.end(), m.group(1)) for m in re.finditer(r"\x60\x60\x60(.*?)\x60\x60\x60", _p, re.S)]
)
def _items_total(body):
    # the receipt's item prices, skipping discount and tax lines
    total = 0.0
    for line in body.splitlines():
        if re.search(r"discount|tax", line, re.I):
            continue
        total += sum(float(x) for x in re.findall(r"\$\s*(\d+\.\d{2})", line))
    return round(total, 2)
def _worked_example(i):
    s, e, body = _blocks[i]
    nxt = _blocks[i + 1][0] if i + 1 < len(_blocks) else len(_p)
    t = re.search(r"total:\s*\$\s*(\d+\.\d{2})", _p[e:nxt], re.I)
    return (bool(t) and len(re.findall(r"\$\s*\d+\.\d{2}", body)) >= 2
            and abs(float(t.group(1)) - _items_total(body)) < 0.005)
`;

export const T_FINAL = String.raw`assert len(_blocks) >= 2 and len(re.findall(r"\d+\.\d{2}", _blocks[-1][2])) >= 2 and all(_blocks[-1][2].strip() != b[2].strip() for b in _blocks[:-1]) and not re.search(r"\d", _p[_blocks[-1][1]:])`;
