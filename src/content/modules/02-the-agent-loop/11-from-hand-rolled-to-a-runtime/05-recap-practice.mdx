    code: `class Agent:
    def __init__(self, client, system_prompt: str, tools: list, max_steps: int = 10):
        self.client = client
        self.system_prompt = system_prompt
        self.tools = {tool.__name__: tool for tool in tools}
        self.max_steps = max_steps

    def invoke(self, user_message: str) -> str:
        messages = [
            {"role": "system", "content": self.system_prompt},
            {"role": "user", "content": user_message},
        ]
        for step_count in range(1, self.max_steps + 1):
            response = self.client.create(messages=messages)
            messages.append({"role": "assistant", "content": response.content})

            tool_calls = [block for block in response.content if block.type == "tool_use"]
            if not tool_calls:
                return "".join(block.text for block in response.content if block.type == "text")

            results = []
            for call in tool_calls:
                tool_function = self.tools[call.name]
                output = tool_function(**call.input)
                results.append({"type": "tool_result", "tool_use_id": call.id, "content": output})
            messages.append({"role": "user", "content": results})
        return f"stopped after reaching the maximum of {self.max_steps} steps without a final answer"`,
