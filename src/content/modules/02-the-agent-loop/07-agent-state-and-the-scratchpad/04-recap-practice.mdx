export const CHECKPOINT_PY = `import json

def serialize_messages(messages: list) -> list:
    serializable = []
    for message in messages:
        content = message["content"]
        if isinstance(content, list):
            content = [block.to_dict() if hasattr(block, "to_dict") else block for block in content]
        serializable.append({"role": message["role"], "content": content})
    return serializable

def save_checkpoint(messages: list, filepath: str) -> None:
    serialized = serialize_messages(messages)
    with open(filepath, "w") as f:
        json.dump(serialized, f, indent=2)

def load_checkpoint(filepath: str) -> list:
    with open(filepath, "r") as f:
        return json.load(f)`;
