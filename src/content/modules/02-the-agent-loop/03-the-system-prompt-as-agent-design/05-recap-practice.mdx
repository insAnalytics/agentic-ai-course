export const PARSE = String.raw`import re
_t = " ".join(system_prompt.lower().split())
_norm = re.sub(r"[ \t]*\n(?![ \t]*(?:\n|[-*]|\d+[.)]))", " ", system_prompt.lower())
_sents = [s.strip() for s in re.split(r"(?<=[.!?])\s+|\n+", _norm) if s.strip()]
_CHK, _CRE = "check_agent_exists", "create_agent_entry"
_chk = [i for i, s in enumerate(_sents) if _CHK in s]
_cre = [i for i, s in enumerate(_sents) if _CRE in s]
_rule = re.compile(r"\b(?:always|never|must not|do not|don't)\b")
_when = re.compile(r"\b(?:whenever|when|if|before|first|prior to|each time|every time|once|after|only|otherwise)\b")
_res = re.compile(r"\b(?:returns?|returned|results?|true|false)\b|already exists?|(?:does(?:n't| not)|not) exist|not found")
_act = re.compile(r"\b(?:proceed|call|inform|tell|say|report|respond|create|register|stop|skip|do not|don't|only|should|must|never|ask|let)\b")
`;

export const T_ROLE = String.raw`assert re.search(r"\byou are (?:a|an|the|our)\b", _t)`;
export const T_RULE = String.raw`assert any(_rule.search(s) and _CHK not in s and _CRE not in s and not re.search(r"\b(?:returns?|returned|results?)\b", s) for s in _sents)`;
export const T_SEQ = String.raw`assert any(
    (_CHK in s and re.search(r"\b(?:before|first|prior to|beforehand)\b", s) and re.search(r"creat|regist|" + _CRE, s))
    or (_CRE in s and re.search(r"\b(?:only if|only when|only after|after|once|unless|if)\b", s) and re.search(_CHK + r"|false|not exist|n't exist|not already|no existing", s))
    or (_CRE in s and re.search(r"\botherwise\b", s))
    or re.search(_CHK + r"[^.!?]*\bthen\b[^.!?]*creat", s)
    for s in _sents)`;
export const T_NOREV = String.raw`assert not re.search(r"(?:call|use|run|invoke)\s+" + _CRE + r"\s+first|" + _CRE + r"\s*,?\s*(?:and\s+)?then\s+(?:call\s+)?" + _CHK, _t)`;
