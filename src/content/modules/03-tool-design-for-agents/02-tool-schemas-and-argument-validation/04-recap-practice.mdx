    code: `from pydantic import BaseModel, field_validator, model_validator

VALID_MODELS = {"claude-sonnet", "claude-haiku", "claude-opus"}

class UpdateModelArgs(BaseModel):
    agent_name: str
    model: str
    fallback_model: str | None = None

    @field_validator("model")
    @classmethod
    def model_must_be_valid(cls, value: str) -> str:
        if value not in VALID_MODELS:
            raise ValueError(f"'{value}' is not a valid model")
        return value

    @model_validator(mode="after")
    def fallback_must_be_valid_and_different(self):
        if self.fallback_model is None:
            return self
        if self.fallback_model not in VALID_MODELS:
            raise ValueError(f"fallback '{self.fallback_model}' is not a valid model")
        if self.fallback_model == self.model:
            raise ValueError("fallback_model must differ from model")
        return self

def update_agent_model(agent_name: str, model: str, registry: dict, fallback_model: str | None = None) -> str:
    try:
        args = UpdateModelArgs(agent_name=agent_name, model=model, fallback_model=fallback_model)
    except Exception as e:
        return f"Error: {str(e)}"
    if args.agent_name not in registry:
        return f"Error: no agent named '{args.agent_name}' exists in the registry"
    registry[args.agent_name]["model"] = args.model
    if args.fallback_model is not None:
        registry[args.agent_name]["fallback_model"] = args.fallback_model
    return f"{args.agent_name} updated to {args.model}"`,
