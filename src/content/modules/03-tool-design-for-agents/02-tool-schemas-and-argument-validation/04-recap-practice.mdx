from pydantic import BaseModel, ValidationError, field_validator, model_validator

ModelName = Literal["claude-sonnet", "claude-haiku", "claude-opus"]

def short_error(e: ValidationError) -> str:
    problems = []
    for error in e.errors():
        field = ".".join(str(part) for part in error["loc"])
        problems.append(f"{field}: {error['msg']}" if field else error["msg"])
    return "; ".join(problems)

class UpdateModelArgs(BaseModel):
    agent_name: str
    model: ModelName
    fallback_model: ModelName | None = None

    @field_validator("agent_name")
    @classmethod
    def name_follows_convention(cls, value: str) -> str:
        if value != value.lower() or not value.endswith("_agent"):
            raise ValueError(f"'{value}' must be lowercase and end in _agent")
        return value

    @model_validator(mode="after")
    def fallback_must_differ(self):
        if self.fallback_model is not None and self.fallback_model == self.model:
            raise ValueError("fallback_model must differ from model")
        return self

def update_agent_model(agent_name: str, model: str, registry: dict, fallback_model: str | None = None) -> str:
    try:
        args = UpdateModelArgs(agent_name=agent_name, model=model, fallback_model=fallback_model)
    except ValidationError as e:
        return f"Error: {short_error(e)}"
    if args.agent_name not in registry:
        return f"Error: no agent named '{args.agent_name}' exists in the registry"
    registry[args.agent_name]["model"] = args.model
    if args.fallback_model is not None:
        registry[args.agent_name]["fallback_model"] = args.fallback_model
    return f"{args.agent_name} updated to {args.model}"`,
    explanation:
      "This composes the whole lesson, and Lesson 1, on one tool. Each rule sits in the place that fits it. The fixed set of models is a Literal, so the allowed values are in the schema the model reads before it calls, and Pydantic rejects anything else with a message listing them (the schema test checks they really are there). A rule a type can't express, the naming convention, is a field_validator. A rule that needs two fields at once, a fallback that differs from the model it backs up, is a model_validator. The registry check handles the world's current state. Every failure comes back as a short observation, trimmed with short_error, and nothing is written to the registry until all of them pass. A field_validator is still the right tool for a set of allowed values when that set isn't fixed, for example models that change at runtime, since a Literal is frozen when the code is written.",
