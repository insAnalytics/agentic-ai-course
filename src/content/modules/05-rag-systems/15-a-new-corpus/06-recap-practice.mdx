def wilson(successes: int, n: int, z: float = 1.96) -> tuple[float, float]:
    """A 95% confidence interval for successes out of n: the Wilson score interval."""
    rate = successes / n
    centre = (rate + z * z / (2 * n)) / (1 + z * z / n)
    spread = z / (1 + z * z / n) * math.sqrt(rate * (1 - rate) / n + z * z / (4 * n * n))
    return centre - spread, centre + spread

export const REPORT_REFERENCE = String.raw`from lib import answerable, sign_test, wilson

def ledger(stages: dict, questions: list[dict], k: int = 5) -> list[dict]:
    """One row per stage, in order: how many questions it answers in the top k and at rank 1,
    and which it gained and lost in the top k against the stage before it."""
    scored = [q for q in questions if q["evidence"]]
    rows, previous = [], None
    for name, search in stages.items():
        top = {q["id"] for q in scored if answerable(search(q, k), q)}
        first = {q["id"] for q in scored if answerable(search(q, 1), q)}
        row = {"stage": name, "top": len(top), "first": len(first), "gained": [], "lost": [], "p": None}
        if previous is not None:
            row["gained"], row["lost"] = sorted(top - previous), sorted(previous - top)
            row["p"] = round(sign_test(len(row["gained"]), len(row["lost"])), 3)
        rows.append(row)
        previous = top
    return rows

def final_check(search, labelled: dict, threshold: float, k: int = 5) -> dict:
    """The finished system, run once on the held-out questions only."""
    runs = [(q, search(q, k)) for q in labelled["held_out"]]
    scored = [(q, results) for q, results in runs if q["evidence"]]
    unanswerable = [results for q, results in runs if not q["evidence"]]
    top = sum(answerable(results, q) for q, results in scored)
    low, high = wilson(top, len(scored))
    return {
        "top": top,
        "of": len(scored),
        "interval": (round(low, 2), round(high, 2)),
        "kept": sum(answerable(results, q) and bool(results) and results[0]["score"] >= threshold for q, results in scored),
        "declined": sum(not results or results[0]["score"] < threshold for results in unanswerable),
        "unanswerable": len(unanswerable),
    }
