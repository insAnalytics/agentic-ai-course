import json
import random
from pathlib import Path

ABLATIONS = Path("/data/eval/ablations")
GROUPS = ("question", "other", "lost write", "planted", "broken result")
# below this many tasks, resampling them can't give a fair interval
MIN_TASKS = 10


def load_results() -> dict:
    """Every condition's per-task trial results, as the lesson's comparisons used them."""
    return json.loads((ABLATIONS / "results.json").read_text(encoding="utf-8"))


def passes(results: dict, condition: str, group: str | None = None, split: str | None = None) -> dict[str, list[bool]]:
    """Each task's trial results under one condition, optionally only the tasks in one group, or in one split."""
    return {task: [row["pass"] for row in rows] for task, rows in results["conditions"][condition].items()
            if (group is None or results["tasks"][task]["group"] == group)
            and (split is None or results["tasks"][task]["split"] == split)}


def paired_difference(a: dict, b: dict, alpha: float = 0.05, repeats: int = 4000,
                      seed: int = 0) -> tuple[float, float, float]:
    """B minus A, task by task, with a (1 - alpha) interval from resampling tasks: the first concept's function, with the
    interval's level as a parameter."""
    tasks = sorted(a.keys() & b.keys())
    differences = [sum(b[t]) / len(b[t]) - sum(a[t]) / len(a[t]) for t in tasks]
    rng = random.Random(seed)
    means = sorted(sum(rng.choices(differences, k=len(differences))) / len(differences) for _ in range(repeats))
    return sum(differences) / len(differences), means[int(alpha / 2 * repeats)], means[int((1 - alpha / 2) * repeats) - 1]
`;

export const REPORT_REFERENCE = String.raw`from lib import GROUPS, MIN_TASKS, load_results, paired_difference, passes


def verdict(low: float, high: float) -> str:
    """What an interval says about a difference."""
    if low > 0:
        return "helps"
    if high < 0:
        return "hurts"
    return "no measurable effect"


def ablation_report(results: dict, conditions: list[str], alpha: float = 0.05) -> list[dict]:
    """Every condition against the baseline on the dev tasks, over all of them and within each group. The interval
    level is corrected for the number of comparisons made (Bonferroni: alpha divided by that number), and a comparison
    over fewer than MIN_TASKS tasks gets the verdict "too few tasks"."""
    comparisons = [(condition, group) for condition in conditions for group in (None, *GROUPS)]
    corrected = alpha / len(comparisons)
    report = []
    for condition, group in comparisons:
        before = passes(results, "baseline", group, split="dev")
        after = passes(results, condition, group, split="dev")
        mean, low, high = paired_difference(before, after, alpha=corrected)
        tasks = len(before.keys() & after.keys())
        report.append({"condition": condition, "group": group or "all tasks", "tasks": tasks, "difference": mean,
                       "low": low, "high": high,
                       "verdict": verdict(low, high) if tasks >= MIN_TASKS else "too few tasks"})
    return report

if __name__ == "__main__":
    for row in ablation_report(load_results(), ["layers", "compaction", "no-labels"]) or []:
        print(f"{row['condition']:<10} {row['group']:<13} {row['tasks']:>3} tasks {row['difference']:+6.1%} "
              f"({row['low']:+6.1%} to {row['high']:+6.1%})  {row['verdict']}")
`;
