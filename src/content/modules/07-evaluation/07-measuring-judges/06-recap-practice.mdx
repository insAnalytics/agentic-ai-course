def rogan_gladen(observed: float, tpr: float, tnr: float) -> float:
    """The pass rate corrected for a judge's errors: (observed + TNR - 1) / (TPR + TNR - 1), kept between 0 and 1.
    A judge with TPR + TNR at or below 1 is no better than chance, and nothing can be corrected with it."""
    if tpr + tnr <= 1:
        raise ValueError(f"TPR + TNR is {tpr + tnr:.2f}: the judge is no better than chance")
    return min(1.0, max(0.0, (observed + tnr - 1) / (tpr + tnr - 1)))


def corrected_interval(observed: float, tpr: float, tnr: float, pass_weights: list[float], fail_weights: list[float],
                       repeats: int = 20000) -> tuple[float, float] | None:
    """A 95% interval for the corrected pass rate. Weighted labels carry less information than their number, so each
    side counts as its effective size, (sum of weights)² / sum of squared weights. TPR and TNR are drawn from Beta
    distributions holding that much evidence (with Jeffreys' half a label each way) and corrected; draws that leave a
    judge no better than chance are skipped. None if fewer than half the draws remain."""
    def effective(weights: list[float]) -> float:
        return sum(weights) ** 2 / sum(w * w for w in weights)

    n_pass, n_fail = effective(pass_weights), effective(fail_weights)
    rng, values = random.Random(0), []
    for _ in range(repeats):
        t = rng.betavariate(tpr * n_pass + 0.5, (1 - tpr) * n_pass + 0.5)
        f = rng.betavariate(tnr * n_fail + 0.5, (1 - tnr) * n_fail + 0.5)
        if t + f > 1:
            values.append(rogan_gladen(observed, t, f))
    if len(values) < repeats // 2:
        return None
    values.sort()
    return values[int(0.025 * len(values))], values[int(0.975 * len(values)) - 1]

def validate(rows: list[dict], population: dict, strata: dict, judge: str, kind: str) -> dict:
    """One judge question's test result: its labelled runs, error rates weighted back to the pool the labels were
    drawn from, observed and corrected pass rates, and why a correction couldn't be made, if it couldn't."""
    usable = [r for r in rows
              if r["kind"] == kind and r["split"] == "test" and not r["excluded"] and r["label"] in ("pass", "fail")]
    drawn = Counter(r["stratum"] for r in usable)
    weight = {stratum: strata[kind][stratum] / n for stratum, n in drawn.items()}
    passes = [(weight[r["stratum"]], r[judge]) for r in usable if r["label"] == "pass"]
    fails = [(weight[r["stratum"]], r[judge]) for r in usable if r["label"] == "fail"]
    counts = population[judge][kind]
    report = {"kind": kind, "labelled": len(usable), "observed": counts["pass"] / (counts["pass"] + counts["fail"]),
              "tpr": sum(w for w, v in passes if v == "pass") / sum(w for w, _ in passes) if passes else None,
              "tnr": sum(w for w, v in fails if v == "fail") / sum(w for w, _ in fails) if fails else None,
              "corrected": None, "interval": None, "reason": None}
    if not passes or not fails:
        report["reason"] = "no labelled passes" if not passes else "no labelled fails"
        return report
    try:
        report["corrected"] = rogan_gladen(report["observed"], report["tpr"], report["tnr"])
    except ValueError:
        report["reason"] = "no better than chance"
        return report
    report["interval"] = corrected_interval(report["observed"], report["tpr"], report["tnr"],
                                            [w for w, _ in passes], [w for w, _ in fails])
    return report
